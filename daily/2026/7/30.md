database.py

```python
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker

# 接続文字列（SQLiteの例。PostgreSQL等の場合は URL を書き換えます）
DATABASE_URL = "sqlite:///./app.db"

# Engine（DBとの低レベルな接続）の作成
engine = create_engine(DATABASE_URL, echo=True)

# SessionFactory（Sessionを発行するクラス）の作成
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
```



models.py

ファイルを分けるのが良いでしょう
```python
from typing import Optional
from sqlalchemy import Text, select
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, Session

class Base(DeclarativeBase):
    pass

# --- モデル定義 ---
class SourceModel(Base):
    __tablename__ = "sources"

    id: Mapped[int] = mapped_column(primary_key=True)
    code: Mapped[str] = mapped_column(Text, unique=True)
    name_ja: Mapped[str] = mapped_column(Text)
    name_en: Mapped[Optional[str]] = mapped_column(Text)
    base_url: Mapped[Optional[str]] = mapped_column(Text)
    license: Mapped[str] = mapped_column(Text)
    redistributable: Mapped[bool] = mapped_column(default=True)

# --- リポジトリ定義 ---
class SourceRepository:
    def __init__(self, session: Session):
        self.session = session

    def get_by_code(self, code: str) -> SourceModel | None:
        stmt = select(SourceModel).where(SourceModel.code == code)
        return self.session.scalars(stmt).first()

    def add(self, source: SourceModel) -> None:
        self.session.add(source)
```

呼び出し方
```python
from database import engine, SessionLocal
from models import Base, SourceModel, SourceRepository

# 1. テーブルが無い場合は自動作成する（初回のみ）
Base.metadata.create_all(bind=engine)

# 2. セッションを開いてリポジトリを使う
def main():
    # with構文を使うことで、処理が終わったら自動でセッションが閉じられます
    with SessionLocal() as session:
        repo = SourceRepository(session)
        
        # データの登録例
        new_source = SourceModel(
            code="SRC001",
            name_ja="オープンデータ1",
            license="CC-BY-4.0"
        )
        session.add(new_source)
        session.commit() # 保存を確定
        
        # データの取得例
        source = repo.get_by_code("SRC001")
        if source:
            print(f"取得成功: {source.name_ja}")

if __name__ == "__main__":
    main()
```


```python
from dataclasses import dataclass
from sqlalchemy import text
from sqlalchemy.orm import Session

# 取得したい結果のデータ構造を定義 (DTO)
@dataclass
class SourceSummaryDTO:
    source_code: str
    source_name: str
    author_name: str
    category_name: str

class SourceReportRepository:
    def __init__(self, session: Session):
        self.session = session

    def get_summary_report(self) -> list[SourceSummaryDTO]:
        query = text("""
            SELECT 
                s.code AS source_code,
                s.name_ja AS source_name,
                u.name AS author_name,
                c.name AS category_name
            FROM sources s
            JOIN users u ON s.created_by_id = u.id
            JOIN categories c ON s.category_id = c.id;
        """)

        result = self.session.execute(query)
        
        # マッピングしてDTOのリストとして返却
        return [
            SourceSummaryDTO(
                source_code=row.source_code,
                source_name=row.source_name,
                author_name=row.author_name,
                category_name=row.category_name
            )
            for row in result
        ]
```
